423template <Key K>
424concept typed_key = requires { typename key_traits<K>::value_type; };
426template <Key K>

434namespace detail
435{
436template <typename T, auto Enc, auto Dec>
437struct WritableTraits
438{
439 using value_type = T;
440 static constexpr bool writable = true;
441 [[nodiscard]] static auto encode(const T & v) noexcept { return Enc(v); }
442 [[nodiscard]] static std::expected<T, KeyError> decode(std::span<const std::byte> v) noexcept
443 {
444 return Dec(v);
445 }
446};
447template <typename T, auto Dec>
448struct ReadOnlyTraits
449{
450 using value_type = T;
451 static constexpr bool writable = false;
452 [[nodiscard]] static std::expected<T, KeyError> decode(std::span<const std::byte> v) noexcept
453 {
454 return Dec(v);
455 }
456};
457[[nodiscard]] inline std::expected<std::string, KeyError> decode_string_owned(
458 std::span<const std::byte> v) noexcept
459{
460 return std::string(decode_string(v));
461}
462} // namespace detail
