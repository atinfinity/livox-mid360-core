116class BoundedQueue
117{
118public:
120 explicit BoundedQueue(std::size_t capacity = 8) : capacity_(capacity) {}
121
123 void push(T && item)
124 {
125 {
126 std::lock_guard lock(mutex_);
127 if (closed_) {
128 return;
129 }
130 if (items_.size() >= capacity_) {
131 items_.pop_front();
132 ++dropped_;
133 }
134 items_.push_back(std::move(item));
135 }
136 cv_.notify_one();
137 }
138
140 [[nodiscard]] std::optional<T> pop(std::chrono::nanoseconds timeout)
141 {
142 std::unique_lock lock(mutex_);
143 cv_.wait_for(lock, timeout, [this] { return closed_ || !items_.empty(); });
144 return take();
145 }
146
148 [[nodiscard]] std::optional<T> try_pop()
149 {
150 std::lock_guard lock(mutex_);
151 return take();
152 }
153
155 void close()
156 {
157 {
158 std::lock_guard lock(mutex_);
159 closed_ = true;
160 }
161 cv_.notify_all();
162 }
163
165 [[nodiscard]] std::uint64_t dropped() const
166 {
167 std::lock_guard lock(mutex_);
168 return dropped_;
169 }
170
171 [[nodiscard]] std::size_t size() const
172 {
173 std::lock_guard lock(mutex_);
174 return items_.size();
175 }
176
177 [[nodiscard]] std::size_t capacity() const noexcept { return capacity_; }
178
179private:
180 std::optional<T> take()
181 {
182 if (items_.empty()) {
183 return std::nullopt;
184 }
185 std::optional<T> out{std::move(items_.front())};
186 items_.pop_front();
187 return out;
188 }
189
190 std::size_t capacity_;
191 mutable std::mutex mutex_;
192 std::condition_variable cv_;
193 std::deque<T> items_;
194 std::uint64_t dropped_ = 0;
195 bool closed_ = false;
196};
