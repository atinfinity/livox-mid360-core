26consteval std::array<std::uint16_t, 256> make_crc16_table()
27{
28 std::array<std::uint16_t, 256> t{};
29 for (std::uint32_t i = 0; i < 256; ++i) {
30 auto c = static_cast<std::uint16_t>(i << 8);
31 for (int k = 0; k < 8; ++k) {
32 const std::uint32_t shifted = static_cast<std::uint32_t>(c) << 1;
33 c = static_cast<std::uint16_t>((c & 0x8000u) ? (shifted ^ 0x1021u) : shifted);
34 }
35 t[i] = c;
36 }
37 return t;
38}

40consteval std::array<std::uint32_t, 256> make_crc32_table()
41{
42 // Reflected form of 0x04C11DB7 is 0xEDB88320.
43 std::array<std::uint32_t, 256> t{};
44 for (std::uint32_t i = 0; i < 256; ++i) {
45 std::uint32_t c = i;
46 for (int k = 0; k < 8; ++k) {
47 c = (c & 1u) ? (0xEDB88320u ^ (c >> 1)) : (c >> 1);
48 }
49 t[i] = c;
50 }
51 return t;
52}

60[[nodiscard]] constexpr std::uint16_t crc16_ccitt_false(
61 std::span<const std::byte> data, std::uint16_t state = 0xFFFF) noexcept
62{

64 const auto idx = static_cast<std::uint8_t>((state >> 8) ^ static_cast<std::uint8_t>(b));
65 state = static_cast<std::uint16_t>((state << 8) ^ detail::kCrc16Table[idx]);

72[[nodiscard]] constexpr std::uint32_t crc32(
73 std::span<const std::byte> data, std::uint32_t state = 0) noexcept
74{
75 std::uint32_t c = state ^ 0xFFFFFFFFu;
76 for (std::byte b : data) {
77 c = detail::kCrc32Table[(c ^ static_cast<std::uint8_t>(b)) & 0xFFu] ^ (c >> 8);
