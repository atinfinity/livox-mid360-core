398 template <Key... Ks>
399 requires(sizeof...(Ks) > 0 && (writable_key<Ks> && ...))
400 std::expected<SetResult, DeviceError> set_many(
401 std::optional<RequestOptions> opts, const key_value_t<Ks> &... values)
402 {
403 auto encoded = std::make_tuple(key_traits<Ks>::encode(values)...);
404 const auto kvs = std::apply(
405 [](const auto &... e) {
406 return std::array<KeyValue, sizeof...(Ks)>{
407 KeyValue{static_cast<std::uint16_t>(Ks), std::span<const std::byte>(e)}...};
408 },
409 encoded);
410 auto ack = configure(kvs, opts);
411 if (!ack) {
412 return std::unexpected(ack.error());
413 }
414 return SetResult{.reboot_required = ack->ret_code == RetCode::kParamRebootEffect};
415 }

429 template <Key... Ks>
430 requires(sizeof...(Ks) > 0 && (typed_key<Ks> && ...))
431 std::expected<std::tuple<key_value_t<Ks>...>, DeviceError> get_many(
432 std::optional<RequestOptions> opts = std::nullopt)
433 {
434 const std::array<Key, sizeof...(Ks)> keys{Ks...};
435 auto res = inquire(keys, opts);
436 if (!res) {
437 return std::unexpected(res.error());
438 }
439 std::optional<DeviceError> failed;
440 std::tuple<key_value_t<Ks>...> out{decode_one<Ks>(*res, failed)...};
441 if (failed) {
442 return std::unexpected(*failed);
443 }
444 return out;
445 }

493 template <Key K>
494 static key_value_t<K> decode_one(const InquireResult & res, std::optional<DeviceError> & failed)
495 {
496 if (const auto raw = res.get(K)) {
497 if (auto v = key_traits<K>::decode(*raw)) {
498 return std::move(*v);
499 }
500 }
501 if (!failed) {
502 failed =
503 DeviceError{.kind = DeviceError::Kind::kDecodeFailed, .session = std::nullopt, .key = K};
504 }
505 return {};
506 }
