# livox-mid360-rerun (issue #147): point clouds in the Rerun viewer. Opt-in with LIVOX_MID360_BUILD_RERUN
# because it pulls the Rerun C++ SDK, and through it Apache Arrow; the library itself stays dependency-free.
# An installed rerun_sdk of this exact version is used when found, else the release zip is fetched.
set(LIVOX_MID360_RERUN_VERSION 0.38.1)
set(BUILD_SHARED_LIBS OFF)  # directory scope: rerun_sdk and Arrow as static libraries, as upstream builds them
include(FetchContent)
fetchcontent_declare(rerun_sdk
  URL https://github.com/rerun-io/rerun/releases/download/${LIVOX_MID360_RERUN_VERSION}/rerun_cpp_sdk.zip
  URL_HASH SHA256=d73b1301c3e28212bb50d8dba59585e4edd1b5b91b98be42379e88969119825a
  SYSTEM
  FIND_PACKAGE_ARGS ${LIVOX_MID360_RERUN_VERSION} EXACT CONFIG)
fetchcontent_makeavailable(rerun_sdk)

# The default viewer layout, generated by make_blueprint.py, compiled in as a byte array.
set(blueprint_rbl ${CMAKE_CURRENT_SOURCE_DIR}/default_blueprint.rbl)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${blueprint_rbl})
file(READ ${blueprint_rbl} blueprint_hex HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," blueprint_bytes "${blueprint_hex}")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/default_blueprint.cpp CONTENT
"// Generated by tools/rerun/CMakeLists.txt from default_blueprint.rbl; do not edit.
#include \"viewer.hpp\"
namespace viewer
{
namespace
{
constexpr unsigned char kBlueprint[] = {@blueprint_bytes@};
}  // namespace
std::span<const std::byte> default_blueprint() { return std::as_bytes(std::span(kBlueprint)); }
}  // namespace viewer
" @ONLY)

add_executable(livox-mid360-rerun main.cpp viewer.cpp live.cpp play.cpp ${CMAKE_CURRENT_BINARY_DIR}/default_blueprint.cpp)
target_include_directories(livox-mid360-rerun PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(livox-mid360-rerun PRIVATE livox::mid360_core rerun_sdk)
livox_mid360_apply_warnings(livox-mid360-rerun)
install(TARGETS livox-mid360-rerun RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

if(LIVOX_MID360_BUILD_TESTS)
  # play of the committed mini.lvx2 and live from the simulator, both into an .rrd file.
  # Shares the simulator's default ports with the example tests, hence the lock.
  add_test(NAME rerun_live_play
    COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/test_rerun.sh $<TARGET_FILE:livox-mid360-rerun>
      ${PROJECT_SOURCE_DIR}/tools/livox_mid360_sim.py ${PROJECT_SOURCE_DIR}/tests/data/mini.lvx2
      ${CMAKE_CURRENT_BINARY_DIR}/test_rerun)
  set_tests_properties(rerun_live_play PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120
    RESOURCE_LOCK sim_default_ports)
endif()
